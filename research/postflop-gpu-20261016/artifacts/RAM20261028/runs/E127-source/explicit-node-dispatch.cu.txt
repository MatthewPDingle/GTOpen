__device__ __forceinline__ void star_action_dispatch(
    u32 n, int p,
    const int* __restrict__ node_player, const int* __restrict__ node_na,
    const u64* __restrict__ node_data_off,
    const u32* __restrict__ node_children_start,
    const u32* __restrict__ children,
    float* regrets_p, float* strat_p,
    const u32* __restrict__ rsrc_p,
    const float* __restrict__ reach_p_buf,
    const long long* __restrict__ lock_off,
    const float* __restrict__ lock_sigma,
    const u32* __restrict__ cfv_slot, float* cfv,
    const float* __restrict__ disc, // [pos, neg, strat] — device-resident so
                                    // captured graphs stay iteration-invariant
    int nh_p, int nh_max)
{
    const int na = node_na[n];
    // Compile-time common action counts keep working arrays in registers.
    // The generic implementation retains arbitrary action-menu support.
    if (na == 2) star_action_impl<2>(n, p, node_player, node_na, node_data_off, node_children_start, children, regrets_p, strat_p, rsrc_p, reach_p_buf, lock_off, lock_sigma, cfv_slot, cfv, disc, nh_p, nh_max);
    else if (na == 3) star_action_impl<3>(n, p, node_player, node_na, node_data_off, node_children_start, children, regrets_p, strat_p, rsrc_p, reach_p_buf, lock_off, lock_sigma, cfv_slot, cfv, disc, nh_p, nh_max);
    else if (na == 4) star_action_impl<4>(n, p, node_player, node_na, node_data_off, node_children_start, children, regrets_p, strat_p, rsrc_p, reach_p_buf, lock_off, lock_sigma, cfv_slot, cfv, disc, nh_p, nh_max);
    else star_action_impl<0>(n, p, node_player, node_na, node_data_off, node_children_start, children, regrets_p, strat_p, rsrc_p, reach_p_buf, lock_off, lock_sigma, cfv_slot, cfv, disc, nh_p, nh_max);
}
