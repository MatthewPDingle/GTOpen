def read(path):
    return json.loads(path.read_bytes())

def sha(path):
    h = hashlib.sha256()
    with path.open('rb') as stream:
        for block in iter(lambda: stream.read(1 << 20), b''):
            h.update(block)
    return h.hexdigest()

def identity(pid, created):
    p = psutil.Process(pid)
    assert abs(p.create_time() - created) < .01
    return dict(pid=pid, created=p.create_time(), parent=p.ppid(), command=p.cmdline())
