def digest(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def read(name):
    path = OUT / name
    inputs[str(path)] = digest(path)
    return json.loads(path.read_bytes())


def require(value, message):
    if not value:
        raise ValueError(message)


def root(desc):
    rows = [n for n in desc['nodes'] if not n['history']]
    require(len(rows) == 1 and rows[0]['player'] == 'BB', 'Unique BB root required')
    return rows[0]


def fourbet(desc):
    rows = [n for n in desc['nodes'] if n['player'] == 'BB'
            and len(n['history']) == 2
            and all(h['action']['kind'] == 'raise' for h in n['history'])]
    require(len(rows) == 1, 'Unique BB four-bet response required')
    return rows[0]


def hand(node, name):
    return next(h for h in node['classes'] if h['hand'] == name)


def call(node, name=None):
    index = next(i for i, a in enumerate(node['actions']) if a['kind'] == 'call')
    probabilities = (hand(node, name)['conditional_hand_action_policy']
                     if name else node['conditional_action_frequencies'])
    return 100 * probabilities[index]
