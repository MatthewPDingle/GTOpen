//! CPU-only canonical state census for a potential public-chance sampler.
//! Does not implement sampling CFR or allocate CUDA. CONTEXT MANIFEST OUTPUT.
use serde_json::{json, Value};
use solver::{Spot, SpotConfig, StreetSizing, TreeConfig, parse_sizes};
use solver::{cards::permute_card, game::Dealt, gpu::stored_capacity_plan};
use solver::preflop::equity::class_label;
use solver::tree::{KIND_ACTION, KIND_CHANCE, SENTINEL};
use std::{collections::BTreeMap, fs::OpenOptions, io::Write};

fn census(spot: &Spot) -> Value {
    // Public keys retain turn/river order. Actions before each deal remain
    // separate information sets; only their allocation byte counts are summed.
    let mut groups: BTreeMap<Vec<u8>, (u64,u64)> = BTreeMap::new();
    let mut todo = vec![(0u32, Dealt::default(), 1u64)];
    let mut action_nodes = 0u64;
    while let Some((index, dealt, multiplicity)) = todo.pop() {
        let node = &spot.tree.nodes[index as usize];
        match node.kind {
            KIND_ACTION => {
                let key = dealt.cards[..dealt.len as usize].to_vec();
                let entry = groups.entry(key).or_insert((0,multiplicity));
                assert_eq!(entry.1,multiplicity);
                entry.0 += node.num_children as u64 * spot.hands[node.player as usize].len() as u64 * 8;
                action_nodes += 1;
                for a in 0..node.num_children as usize {
                    todo.push((spot.tree.children[node.children_start as usize+a],dealt,multiplicity));
                }
            }
            KIND_CHANCE => {
                let perms = spot.perms_fixing(&dealt);
                let mut orbits: BTreeMap<u8,u64> = BTreeMap::new();
                for c in 0..52u8 {
                    let child = spot.tree.children[node.children_start as usize+c as usize];
                    if child == SENTINEL || dealt.contains(c) {continue;}
                    let rep = perms.iter().map(|&k|permute_card(c,&spot.suit_perms[k])).min().unwrap();
                    *orbits.entry(rep).or_default() += 1;
                }
                assert_eq!(orbits.values().sum::<u64>(),49-dealt.len as u64);
                for (rep,count) in orbits {
                    todo.push((spot.tree.children[node.children_start as usize+rep as usize],
                        dealt.push(rep),multiplicity*count));
                }
            }
            _ => {}
        }
    }
    let mut hist: BTreeMap<(usize,u64),(u64,u64)> = BTreeMap::new();
    for (key,(bytes,multiplicity)) in &groups {
        let h = hist.entry((key.len(),*multiplicity)).or_default();
        h.0 += 1; h.1 += bytes;
    }
    let plan = stored_capacity_plan(spot);
    let total = groups.values().map(|g|g.0).sum::<u64>();
    assert_eq!(total,plan["canonical_state_bytes"].as_u64().unwrap(),"canonical planner mismatch");
    assert_eq!(action_nodes,plan["canonical_actions"].as_u64().unwrap());
    let rows: Vec<_> = hist.iter().map(|(&(depth,mult),&(count,bytes))|json!({
        "depth":depth,"orbit_multiplicity":mult,"public_groups":count,"state_bytes":bytes,
        "physical_public_prefixes": ([1,49,49*48][depth])
    })).collect();
    for depth in 0..3 {
        let prefixes: u64 = hist.iter().filter(|((d,_),_)|*d==depth).map(|((_,m),(n,_))|m*n).sum();
        assert_eq!(prefixes,[1,49,49*48][depth]);
    }
    json!({"canonical_state_bytes":total,"canonical_actions":action_nodes,
        "public_group_count":groups.len(),"histogram":rows})
}

fn main() -> Result<(),Box<dyn std::error::Error>> {
    let args: Vec<_> = std::env::args().skip(1).collect(); assert_eq!(args.len(),3);
    assert!(!std::path::Path::new(&args[2]).exists(),"preserve evidence");
    let context:Value=serde_json::from_slice(&std::fs::read(&args[0])?)?;
    let manifest:Value=serde_json::from_slice(&std::fs::read(&args[1])?)?;
    assert_eq!(context["schema"],"hu-context-v1");
    let ranges:Vec<String>=(0..2).map(|p| {
        let weights:Vec<f64>=(0..169).map(|c|context["incoming_class_mass"][p][c].as_f64().unwrap()/
            if c/13==c%13 {6.} else if c/13>c%13 {4.} else {12.}).collect();
        let max=weights.iter().copied().fold(0.,f64::max);assert!(max>0.);
        (0..169).filter(|&c|weights[c]/max>=1e-5).map(class_label).collect::<Vec<_>>().join(",")
    }).collect();
    let menu=manifest["bet_menu"].as_str().unwrap();
    let sizing=StreetSizing {bet:parse_sizes(menu)?,raise:parse_sizes("100")?,donk:parse_sizes(menu)?};
    let mut rows=Vec::new();
    for board in manifest["boards"].as_array().unwrap() {
        for (leaf,node) in context["nodes"].as_array().unwrap().iter().enumerate() {
            if node["leaf"]["type"]!="postflop" {continue;}
            let spot=Spot::new_with_limit(SpotConfig {
                board:board["board"].as_str().unwrap().into(),range_oop:ranges[0].clone(),range_ip:ranges[1].clone(),
                tree:TreeConfig {starting_pot:node["leaf"]["starting_pot"].as_f64().unwrap(),
                    effective_stack:node["leaf"]["effective_stack"].as_f64().unwrap(),
                    rake_pct:context["rake_fraction"].as_f64().unwrap(),rake_cap:context["rake_cap"].as_f64().unwrap(),
                    max_raises:1,oop:[sizing.clone(),sizing.clone(),sizing.clone()],
                    ip:[sizing.clone(),sizing.clone(),sizing.clone()],..Default::default()}
            },Some(2_000_000))?;
            let mut row=census(&spot);row["board"]=board["board"].clone();row["preflop_leaf"]=json!(leaf);
            println!("PUBLIC_CENSUS {} leaf={} bytes={}",board["board"],leaf,row["canonical_state_bytes"]);
            rows.push(row);
        }
    }
    let result=json!({"rows":rows,"manifest":manifest,"device_allocated":false,"strategies_evaluated":false,
        "note":"Canonical allocation census only. Sampling all public outcomes retains the game in expectation with correct importance weights, but no sampler is implemented or validated here. Lazy storage eventually approaches the full forest."});
    OpenOptions::new().create_new(true).write(true).open(&args[2])?.write_all(&serde_json::to_vec_pretty(&result)?)?;
    Ok(())
}
