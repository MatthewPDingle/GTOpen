"""Synthetic preparation controls; no retained scientific weights or CUDA use."""
import ast
from pathlib import Path
import unittest
import numpy as np
import stack_board_boundary_diagnostic_v1 as diagnostic


class BoundaryDiagnosticControls(unittest.TestCase):
    def test_chunk_geometry_matches_full_original_pass(self):
        # A tiny layout exercises state crossings, board crossings, exact batch
        # endpoints and the final partial batch against an exhaustive oracle.
        for states in (1, 2, 5):
            for holdings in (1, 3, 11):
                for batch in (2, 6, 17):
                    boards, selected = 4, (0, 3)
                    actual = diagnostic.target_chunks(selected, boards, states, holdings, batch)
                    expected = []
                    seen = {}
                    total = boards * holdings * states
                    for lo in range(0, total, batch):
                        hi = min(lo + batch, total)
                        target = [i for i in range(lo, hi) if (i // holdings) % boards in selected]
                        if target:
                            expected.append((lo, hi))
                            for i in target:
                                seen[i] = seen.get(i, 0) + 1
                    self.assertEqual(actual, expected)
                    self.assertEqual(set(seen), {i for i in range(total) if (i // holdings) % boards in selected})
                    self.assertTrue(all(count == 1 for count in seen.values()))

    def test_original_geometry_has_exact_target_coverage(self):
        d = diagnostic
        chunks = d.target_chunks(d.TARGET_BOARDS, d.BOARDS, d.STATES, d.HOLDINGS, d.BATCH)
        nc = d.BOARDS * d.HOLDINGS
        seen = np.zeros((d.STATES, len(d.TARGET_BOARDS), d.HOLDINGS), dtype=np.int8)
        for lo, hi in chunks:
            self.assertEqual(lo % d.BATCH, 0)
            self.assertEqual(hi - lo, min(d.BATCH, nc * d.STATES - lo))
            ids = np.arange(lo, hi)
            for index, board in enumerate(d.TARGET_BOARDS):
                mask = (ids % nc) // d.HOLDINGS == board
                seen[ids[mask] // nc, index, ids[mask] % d.HOLDINGS] += 1
        self.assertTrue(np.all(seen == 1))
        self.assertLess(sum(hi - lo for lo, hi in chunks), nc * d.STATES)

    def test_invalid_targets_are_rejected(self):
        for targets in ((), (0, 0), (-1,), (4,), (True,)):
            with self.assertRaises(ValueError):
                diagnostic.target_chunks(targets, 4, 2, 3, 8)

    def test_score_rule_matches_original_for_boundary_cases(self):
        # Extract only the original pure probability function: importing its
        # training module would needlessly import torch for this CPU control.
        source = Path('T:/Dev/GTOpen/tools/research/stack_price_learning_fit_v1.py')
        tree = ast.parse(source.read_text(encoding='utf-8'))
        function = next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == 'probabilities')
        namespace = {'np': np}
        exec(compile(ast.Module(body=[function], type_ignores=[]), str(source), 'exec'), namespace)
        around = np.nextafter(np.float32(1), np.float32(2))
        tiny = np.nextafter(np.float32(0), np.float32(1))
        raw = np.asarray([[0, 0], [-1, -1], [-2, -1], [-1, -2], [1, 1],
                          [1, around], [around, 1], [tiny, 0], [0, tiny],
                          [-tiny, 0], [0, -tiny], [-0., 0.], [1, -1]], dtype=np.float32)
        got = diagnostic.call_probability(raw)
        self.assertTrue(np.array_equal(got, namespace['probabilities'](raw)))
        self.assertTrue(np.array_equal(got[:5], [0, 0, 1, 0, .5]))
        self.assertGreater(got[5], .5)
        self.assertLess(got[6], .5)
        with self.assertRaises(ValueError):
            diagnostic.call_probability(np.asarray([[0, np.nan]]))

    def test_classification_change_does_not_imply_large_cash_change(self):
        n = diagnostic.HOLDINGS * 2
        cpu = np.ones((1, n, 2), dtype=np.float32)
        gpu = cpu.copy()
        gpu[0, 0, 0] = np.nextafter(np.float32(1), np.float32(2))
        economics = np.asarray([[40, 10, 1, 0, .1]], dtype=float)
        summary, arrays = diagnostic.summarize(cpu, gpu, np.full(n, .2), economics)
        self.assertEqual(summary['disagreement_count'], 1)
        self.assertEqual(summary['boards'][0]['cpu_errors'], 0)
        self.assertEqual(summary['boards'][0]['gpu_errors'], 1)
        self.assertLess(summary['maximum_cash_loss_difference_bb'], 1e-6)
        self.assertEqual(np.count_nonzero(arrays['disagreement']), 1)

    def test_nonpositive_order_crossing_is_preserved(self):
        n = diagnostic.HOLDINGS * 2
        cpu = -np.ones((1, n, 2), dtype=np.float32)
        gpu = cpu.copy()
        gpu[0, 0, 1] = np.nextafter(np.float32(-1), np.float32(0))
        economics = np.asarray([[40, 10, 1, 0, .1]], dtype=float)
        summary, arrays = diagnostic.summarize(cpu, gpu, np.full(n, .2), economics)
        self.assertEqual(summary['disagreement_count'], 1)
        self.assertEqual(summary['maximum_probability_difference'], 1.)
        self.assertEqual(summary['maximum_cash_loss_difference_bb'], 1.)
        self.assertEqual(arrays['cpu_probability'][0, 0], 0.)
        self.assertEqual(arrays['gpu_probability'][0, 0], 1.)


if __name__ == '__main__':
    unittest.main(verbosity=2)
